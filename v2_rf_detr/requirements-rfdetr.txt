# RF-DETR training stack for projeto_placentas.
#
# Windows / conda trap: `conda install pytorch=2.5 ...` can resolve to **CPU-only**
# builds (py3.10_cpu_0) and flip pytorch-mutex to **cpu**, breaking GPU and sometimes
# leaving transformers imports inconsistent. Prefer pip CUDA wheels below.
#
# Recover GPU + satisfy transformers (torch >= 2.4):
#   conda activate projeto_placentas
#   conda remove pytorch torchvision torchaudio pytorch-cuda pytorch-mutex --yes
#   python -m pip install --upgrade torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu121
#   python -m pip install --upgrade "transformers>=5.1,<6"
#   python -m pip install "rfdetr[train]>=1.6.2"
#
# Verify:
#   python -c "import torch; print(torch.__version__, torch.cuda.is_available())"
#   python -c "from transformers import BackboneConfigMixin; from rfdetr import RFDETRSegSmall; print('ok')"

torch>=2.4,<3
transformers>=5.1,<6
rfdetr[train]>=1.6.2
